cmake_minimum_required(VERSION 3.24)
project(CrashFixture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENGINE_BUILD_RUNTIME ON)
set(ENGINE_BUILD_EDITOR ON)
option(CRASH_FIXTURE_EDITOR_TESTS "Build targeted editor checks" ON)
set(ENGINE_BUILD_EDITOR_TESTS ${CRASH_FIXTURE_EDITOR_TESTS})
add_subdirectory("${ENGINE_SOURCE}" Engine)
engine_add_game(CrashFixture SOURCES Game.cpp)
engine_add_editor_worker(CrashFixtureWorker SOURCES Worker.cpp)
set_target_properties(CrashFixture PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/$<CONFIG>")
